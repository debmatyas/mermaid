        CDN[Cloudflare CDN / Edge Cache]
        WAF[Web Application Firewall & DDoS]
        APIGateway[API Gateway / Kong]
    end

    subgraph SecurityLayer ["2. Biztonság & Identitás"]
        AuthService[Auth / IAM Service - Keycloak]
        TokenStore[(Redis Token Store)]
        OAuthProvider[OAuth2 / OIDC Google/Apple]
    end

    %% 3. ALKALMAZÁS / MIKROSZOLGÁLTATÁSOK
    subgraph Microservices ["3. Mikroszolgáltatások Magja"]
        direction TB
        subgraph UserDomain ["Felhasználói Domén"]
            UserProfile[User Profile Service]
            UserPref[User Preferences Service]
        end

        subgraph CatalogDomain ["Katalógus Domén"]
            ProductCatalog[Product Catalog Service]
            SearchService[ElasticSearch Indexer]
            ReviewService[Product Reviews & Ratings]
        end

        subgraph OrderDomain ["Rendelési & Fizetési Domén"]
            OrderService[Order Orchestrator Service]
            CartService[Shopping Cart Service]
            CheckoutService[Checkout Flow Service]
            PaymentProcessor[Payment Processing Service]
        end

        subgraph LogisticsDomain ["Logisztika & Raktár Domén"]
            InventoryService[Inventory Stock Service]
            WarehouseManager[Warehouse Management API]
            ShippingService[Shipping & Fulfillment]
        end

        subgraph SupportDomain ["Értesítési & Egyéb Szolgáltatások"]
            NotificationService[Notification Engine]
            RecommendationAI[AI Recommendation Engine]
            AnalyticsCollector[Real-time Analytics Collector]
        end
    end

    %% 4. ÜZENETKEZELÉS & STREAMING
    subgraph MessagingLayer ["4. Eseményvezérelt Réteg (Kafka)"]
        KafkaCluster((Apache Kafka Cluster))
        SchemaRegistry[Confluent Schema Registry]
    end

    %% 5. ADATBÁZISOK ÉS TÁROLÓK

    end

    end
